cmake_minimum_required(VERSION 3.8)
project(coal_scene_query)
find_package(ament_cmake REQUIRED)
add_library(coal_query_math SHARED src/query_math.cpp)
target_compile_features(coal_query_math PRIVATE cxx_std_17)
target_compile_options(coal_query_math PRIVATE -O3 -Wall -Wextra)
install(TARGETS coal_query_math LIBRARY DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/scene_query.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/geometry_engine.py scripts/collision_loader.py DESTINATION lib/${PROJECT_NAME})
ament_package()
